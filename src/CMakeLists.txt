SET(SOURCES main.cpp cCommandLine.cpp)
SET(MODULE_LIBRARIES)
IF(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    LIST(APPEND SOURCES cRecorder.cpp cWaitSampler.cpp cCpuSampler.cpp cProcessCodeSource.cpp)
    SET(MODULE_LIBRARIES ssr-process ssr-wait ssr-stack ssr-cpu ssr-profile)
    IF(TARGET ssr-bpf)
        LIST(APPEND MODULE_LIBRARIES ssr-bpf)
    ENDIF()
ENDIF()

ADD_EXECUTABLE(${PROJECT_NAME} ${SOURCES})
TARGET_LINK_LIBRARIES(${PROJECT_NAME} PRIVATE ssr-warnings ${MODULE_LIBRARIES})
TARGET_COMPILE_DEFINITIONS(${PROJECT_NAME} PRIVATE STACK_SAMPLER_VERSION="${PROJECT_VERSION}" STACK_SAMPLER_NAME="${PROJECT_NAME}"
    STACK_SAMPLER_COMMIT="${PROJECT_COMMIT}")
INSTALL(TARGETS ${PROJECT_NAME} RUNTIME DESTINATION bin)
